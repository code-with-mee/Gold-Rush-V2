2How day and night work

Press N (or the sun/moon button) and the level turns from day to night in 1.5 seconds. Nothing is loaded. One number does the job:

  • DayNight.Value goes from 0 (day) to 1 (night).
  • Every frame, DayNight mixes the day colour and the night colour with this number: sky, fog, sun, ambient light, bloom, vignette.
  • Other objects register (sign up) with DayNight:
    • a material with a day colour and a night colour (floor, walls, blocks),
    • a night light that is off by day and fades in at night (lamps, the hero's lantern, monster glows).
ScriptJob
DayNightThe blend. Mixes all colours and lights.
NightLightPut it on a Light: it only shines at night.
DayNightTintPut it on the level root: a list of materials with day/night colours.
DayNightPaletteOne asset with the colours for all levels, so prefabs change colour everywhere.
EnvironmentSetupMakes the sky, fog, sun and post-processing, then creates DayNight.
DayNightButtonThe round sun/moon button (and the N key).

3Script: DayNight

Assets/GoldRush/Scripts/World/DayNight.cs – replace the whole file.

using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Rendering.Universal;

namespace GoldRush
{
    /// <summary>
    /// Real-time day/night blend (0 = day, 1 = night). Everything that should look different at night
    /// registers here: material colours (floor, walls), lights that only shine at night (lanterns,
    /// monster glows, lamps). Call <see cref="Toggle"/> to switch smoothly at any time.
    /// </summary>
    public class DayNight : MonoBehaviour
    {
        public static DayNight I { get; private set; }

        /// <summary>Current blend value, 0 = full day, 1 = full night.</summary>
        public static float Value => I ? I.value : 0f;
        /// <summary>True when heading to (or at) night.</summary>
        public static bool IsNight => I && I.target > 0.5f;
        public static event System.Action<bool> Changed;

        [Tooltip("Seconds for a full day <-> night transition.")]
        public float transitionTime = 1.5f;

        float value, target;
        Light sun;
        Camera cam;
        Material sky;
        Bloom bloom;
        Vignette vignette;

        struct Tint { public Material mat; public Color day, night; }
        struct NightLight { public Light light; public float intensity; }
        readonly List<Tint> tints = new List<Tint>();
        readonly List<NightLight> nightLights = new List<NightLight>();

        // registrations made before DayNight exists (e.g. scene objects whose Start runs first)
        static readonly List<Tint> pendingTints = new List<Tint>();
        static readonly List<NightLight> pendingLights = new List<NightLight>();

        // ---- presets
        static readonly Color DaySky = new Color(0.78f, 0.82f, 0.9f), NightSky = new Color(0.2f, 0.22f, 0.42f);
        static readonly Color DayEquator = new Color(0.62f, 0.65f, 0.72f), NightEquator = new Color(0.12f, 0.13f, 0.27f);
        static readonly Color DayGround = new Color(0.4f, 0.4f, 0.45f), NightGround = new Color(0.05f, 0.05f, 0.1f);
        static readonly Color DayFog = new Color(0.72f, 0.78f, 0.88f), NightFog = new Color(0.05f, 0.06f, 0.14f);
        static readonly Color DaySun = new Color(1f, 0.95f, 0.86f), NightSun = new Color(0.5f, 0.6f, 1f);
        static readonly Quaternion DaySunRot = Quaternion.Euler(50f, 35f, 0f), NightSunRot = Quaternion.Euler(40f, -35f, 0f);

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
        static void ResetStatics() { I = null; Changed = null; pendingTints.Clear(); pendingLights.Clear(); }

        public static DayNight Create(GameObject host, Camera cam, Light sun, Material skyboxAsset, Bloom bloom, Vignette vignette, bool night)
        {
            var d = host.AddComponent<DayNight>();
            I = d;
            d.cam = cam;
            d.sun = sun;
            d.bloom = bloom;
            d.vignette = vignette;
            if (skyboxAsset)
            {
                d.sky = new Material(skyboxAsset) { name = "Sky (runtime)" };
                RenderSettings.skybox = d.sky;
            }
            night = Save.Data.night; // the player's saved choice wins over the scene default
            d.value = d.target = night ? 1f : 0f;
            d.tints.AddRange(pendingTints);
            d.nightLights.AddRange(pendingLights);
            pendingTints.Clear();
            pendingLights.Clear();
            d.ApplyNow();
            return d;
        }

        void OnDestroy() { if (I == this) I = null; }

        // ------------------------------------------------------------------ public API

        public static void Toggle() { if (I) Set(!IsNight); }

        public static void Set(bool night, bool instant = false)
        {
            if (!I) return;
            I.target = night ? 1f : 0f;
            if (Save.Data.night != night) { Save.Data.night = night; Save.Persist(); } // remembered everywhere
            if (instant) { I.value = I.target; I.ApplyNow(); }
            Changed?.Invoke(night);
        }

        /// <summary>Material whose _BaseColor blends between day and night colours.</summary>
        public static void RegisterTint(Material m, Color day, Color night)
        {
            if (!m) return;
            var tint = new Tint { mat = m, day = day, night = night };
            if (!I) { pendingTints.Add(tint); m.SetColor("_BaseColor", day); return; }
            I.tints.Add(tint);
            m.SetColor("_BaseColor", Color.Lerp(day, night, Smooth(I.value)));
        }

        /// <summary>Light that fades in at night (to the given intensity) and switches off by day.</summary>
        public static void RegisterNightLight(Light l, float nightIntensity = -1f)
        {
            if (!l) return;
            float k = nightIntensity >= 0 ? nightIntensity : l.intensity;
            var nl = new NightLight { light = l, intensity = k };
            if (!I) { pendingLights.Add(nl); return; }
            I.nightLights.Add(nl);
            I.ApplyLight(I.nightLights[I.nightLights.Count - 1], Smooth(I.value));
        }

        // ------------------------------------------------------------------ blending

        static float Smooth(float t) => t * t * (3f - 2f * t);

        void Update()
        {
            if (Mathf.Approximately(value, target)) return;
            value = Mathf.MoveTowards(value, target, Time.unscaledDeltaTime / Mathf.Max(0.05f, transitionTime));
            ApplyNow();
        }

        void ApplyNow()
        {
            float t = Smooth(value);
            RenderSettings.ambientMode = UnityEngine.Rendering.AmbientMode.Trilight;
            RenderSettings.ambientSkyColor = Color.Lerp(DaySky, NightSky, t);
            RenderSettings.ambientEquatorColor = Color.Lerp(DayEquator, NightEquator, t);
            RenderSettings.ambientGroundColor = Color.Lerp(DayGround, NightGround, t);
            RenderSettings.fog = true;
            RenderSettings.fogMode = FogMode.Linear;
            RenderSettings.fogColor = Color.Lerp(DayFog, NightFog, t);
            RenderSettings.fogStartDistance = Mathf.Lerp(40f, 18f, t);
            RenderSettings.fogEndDistance = Mathf.Lerp(140f, 75f, t);

            if (sun)
            {
                sun.color = Color.Lerp(DaySun, NightSun, t);
                sun.intensity = Mathf.Lerp(1.35f, 0.45f, t);
                sun.transform.rotation = Quaternion.Slerp(DaySunRot, NightSunRot, t);
            }
            if (sky)
            {
                sky.SetFloat("_Exposure", Mathf.Lerp(1.1f, 0.06f, t));
                sky.SetFloat("_AtmosphereThickness", Mathf.Lerp(0.7f, 0.4f, t));
                sky.SetColor("_SkyTint", Color.Lerp(new Color(0.5f, 0.55f, 0.65f), new Color(0.2f, 0.22f, 0.5f), t));
                sky.SetColor("_GroundColor", Color.Lerp(new Color(0.62f, 0.66f, 0.74f), new Color(0.05f, 0.06f, 0.14f), t));
                sky.SetFloat("_SunSize", 0.035f);
            }
            if (cam)
            {
                cam.backgroundColor = Color.Lerp(DayFog, NightFog, t);
                cam.clearFlags = sky ? CameraClearFlags.Skybox : CameraClearFlags.SolidColor;
            }
            if (bloom != null)
            {
                bloom.intensity.Override(Mathf.Lerp(0.6f, 1.1f, t));
                bloom.threshold.Override(Mathf.Lerp(0.95f, 0.8f, t));
            }
            if (vignette != null) vignette.intensity.Override(Mathf.Lerp(0.25f, 0.38f, t));

            for (int i = tints.Count - 1; i >= 0; i--)
            {
                if (!tints[i].mat) { tints.RemoveAt(i); continue; }
                tints[i].mat.SetColor("_BaseColor", Color.Lerp(tints[i].day, tints[i].night, t));
            }
            for (int i = nightLights.Count - 1; i >= 0; i--)
            {
                if (!nightLights[i].light) { nightLights.RemoveAt(i); continue; }
                ApplyLight(nightLights[i], t);
            }
        }

        void ApplyLight(NightLight n, float t)
        {
            n.light.intensity = n.intensity * t;
            n.light.enabled = t > 0.01f;
        }
    }
}
  • value moves toward target in Update. Toggle() only changes target, so the change is smooth.
  • Smooth(t) makes the blend start and end slowly (no hard jump).
  • Color.Lerp(day, night, t) mixes two colours: t = 0 gives the day colour, t = 1 the night colour.
  • RegisterTint and RegisterNightLight add objects to the lists. If DayNight does not exist yet, they wait in the pending lists.
  • Changed is an event: the day/night button listens to it to swap its icon.
  • Save.Data.night is ONE day/night setting for the menu and every level. When a scene starts, the saved choice wins over the scene default; Set() (used by Toggle()) saves a new choice with Save.Persist().
  • It uses Time.unscaledDeltaTime, so it still works when the game is paused.

4Script: NightLight

Assets/GoldRush/Scripts/World/NightLight.cs – replace the whole file (or create it).

using UnityEngine;

namespace GoldRush
{
    /// <summary>Put this on any Light: it fades in at night and switches off during the day.</summary>
    [RequireComponent(typeof(Light))]
    public class NightLight : MonoBehaviour
    {
        [Tooltip("Brightness of the light at full night.")]
        public float nightIntensity = 2f;

        void Start()
        {
            if (Application.isPlaying) DayNight.RegisterNightLight(GetComponent<Light>(), nightIntensity);
        }

        void Reset() { nightIntensity = GetComponent<Light>().intensity; }
    }
}
  • In Start it registers its Light with DayNight. By day the light is off. At night it fades in to nightIntensity.
  • Reset() runs when you add the component: it copies the light's current intensity.

Add it to the lamp prefab:

  1. Open Prefabs/Level/LampPost in prefab mode (double-click).
  2. Select the child LampLight (Point light, colour FFE0B3, range 9).
  3. Add Component → Night Light. Set Night Intensity to 2.4.
  4. Save the prefab. All lamps in all levels now turn on at night.

5Scripts: DayNightTint and DayNightPalette

Assets/GoldRush/Scripts/World/DayNightTint.cs – replace the whole file (or create it).

using System.Collections.Generic;
using UnityEngine;

namespace GoldRush
{
    /// <summary>Lists materials whose colour changes between day and night (e.g. floor: light grey by day,
    /// navy blue by night). Put one on the level root and edit the colours in the Inspector.</summary>
    public class DayNightTint : MonoBehaviour
    {
        [System.Serializable]
        public class Entry
        {
            public Material material;
            public Color day = Color.white;
            public Color night = Color.gray;
        }

        public List<Entry> entries = new List<Entry>();

        public void Add(Material m, Color day, Color night)
        {
            if (!m) return;
            foreach (var e in entries) if (e.material == m) { e.day = day; e.night = night; return; }
            entries.Add(new Entry { material = m, day = day, night = night });
        }

        void Start()
        {
            if (!Application.isPlaying) return;
            // Tint runtime COPIES: changing a material asset in Play mode would save the change into the
            // project. Swap the copies onto every renderer in the scene that uses an original.
            var copies = new Dictionary<Material, Material>();
            var all = new List<Entry>(entries);
            var palette = DayNightPalette.Load();
            if (palette) all.AddRange(palette.entries);
            foreach (var e in all)
            {
                if (!e.material || copies.ContainsKey(e.material)) continue;
                var copy = new Material(e.material) { name = e.material.name + " (day/night)" };
                copies[e.material] = copy;
                DayNight.RegisterTint(copy, e.day, e.night);
            }
            foreach (var r in FindObjectsByType<Renderer>())
            {
                var mats = r.sharedMaterials;
                bool changed = false;
                for (int i = 0; i < mats.Length; i++)
                    if (mats[i] && copies.TryGetValue(mats[i], out var c)) { mats[i] = c; changed = true; }
                if (changed) r.sharedMaterials = mats;
            }
        }
    }
}

Assets/GoldRush/Scripts/World/DayNightPalette.cs – replace the whole file (or create it). You imported both files in phase 7; check that they are the same as this code.

using System.Collections.Generic;
using UnityEngine;

namespace GoldRush
{
    /// <summary>
    /// Project-wide list of block materials and their day/night colours
    /// (Assets/GoldRush/Resources/DayNightPalette.asset). Every DayNightTint also applies these,
    /// so prefabs dragged into a level change colour at night automatically.
    /// </summary>
    [CreateAssetMenu(fileName = "DayNightPalette", menuName = "Gold Rush/Day Night Palette")]
    public class DayNightPalette : ScriptableObject
    {
        public List<DayNightTint.Entry> entries = new List<DayNightTint.Entry>();

        public void Add(Material m, Color day, Color night)
        {
            if (!m) return;
            foreach (var e in entries) if (e.material == m) { e.day = day; e.night = night; return; }
            entries.Add(new DayNightTint.Entry { material = m, day = day, night = night });
        }

        public static DayNightPalette Load() => Resources.Load<DayNightPalette>("DayNightPalette");
    }
}
  • DayNightTint makes a copy of each material. If it changed the real material asset in Play mode, Unity would save the night colour into your project.
  • Then it puts the copy on every Renderer in the scene that used the original.
  • DayNightPalette.Load() reads the asset Resources/DayNightPalette. Its colours are added to every level, so you only enter them once.
  • [CreateAssetMenu] adds the palette to the Assets → Create menu.

7Script: EnvironmentSetup

Assets/GoldRush/Scripts/World/EnvironmentSetup.cs – replace the whole file.

using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

namespace GoldRush
{
    /// <summary>Lighting moods (sunset day / moonlit night), fog, skybox and URP post-processing.</summary>
    public static class EnvironmentSetup
    {
        /// <summary>The global vignette override (gameplay may pulse it red on hit).</summary>
        public static Vignette Vignette { get; private set; }

        /// <summary>True when the scene is (heading to) night. See <see cref="DayNight"/>.</summary>
        public static bool IsNight => DayNight.IsNight;

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
        static void ResetStatics() { Vignette = null; }

        public static void Apply(Camera cam, bool menu, bool night = false)
        {
            // ---- sky / ambient / fog
            var assets = GameAssets.I;
            if (assets != null && assets.skybox != null) RenderSettings.skybox = assets.skybox;

            RenderSettings.ambientMode = AmbientMode.Trilight;
            RenderSettings.ambientSkyColor = new Color(0.8f, 0.7f, 0.62f);
            RenderSettings.ambientEquatorColor = new Color(0.74f, 0.56f, 0.42f);
            RenderSettings.ambientGroundColor = new Color(0.46f, 0.33f, 0.24f);

            RenderSettings.fog = true;
            RenderSettings.fogMode = FogMode.Linear;
            RenderSettings.fogColor = new Color(0.96f, 0.79f, 0.6f);
            RenderSettings.fogStartDistance = menu ? 25f : 35f;
            RenderSettings.fogEndDistance = menu ? 80f : 120f;

            // ---- sun
            Light sun = null;
            var lights = Object.FindObjectsByType<Light>();
            for (int i = 0; i < lights.Length; i++)
                if (lights[i].type == LightType.Directional) { sun = lights[i]; break; }
            if (sun == null)
            {
                var go = new GameObject("Sun");
                sun = go.AddComponent<Light>();
                sun.type = LightType.Directional;
                sun.color = new Color(1f, 0.87f, 0.68f);
                sun.intensity = 1.5f;
                sun.shadows = LightShadows.Soft;
                sun.shadowStrength = 0.75f;
                sun.shadowBias = 0.05f;
                sun.shadowNormalBias = 0.4f;
                // low sunset sun behind/left of the default camera direction (+Z)
                go.transform.rotation = Quaternion.Euler(30f, 35f, 0f);
            }
            RenderSettings.sun = sun;
            DynamicGI.UpdateEnvironment();

            // ---- post-processing volume
            var vgo = new GameObject("PostFX");
            var vol = vgo.AddComponent<Volume>();
            vol.isGlobal = true;
            vol.priority = 10f;
            var profile = ScriptableObject.CreateInstance<VolumeProfile>();
            profile.name = "GoldRushPost";

            var bloom = profile.Add<Bloom>(true);
            bloom.intensity.Override(night ? 1.1f : 0.7f);
            bloom.threshold.Override(night ? 0.8f : 0.95f);
            bloom.scatter.Override(0.65f);

            var vig = profile.Add<Vignette>(true);
            vig.intensity.Override(night ? 0.38f : 0.28f);
            vig.smoothness.Override(0.45f);
            vig.color.Override(Color.black);

            var tone = profile.Add<Tonemapping>(true);
            tone.mode.Override(TonemappingMode.Neutral);

            var ca = profile.Add<ColorAdjustments>(true);
            ca.contrast.Override(8f);
            ca.saturation.Override(12f);
            ca.postExposure.Override(0.05f);

            vol.sharedProfile = profile;
            Vignette = vig;

            // ---- real-time day/night blend (colours, fog, sun, sky, bloom, night lights)
            DayNight.Create(vgo, cam, sun, assets != null ? assets.skybox : null, bloom, vig, night);

            // ---- camera
            if (cam != null)
            {
                cam.allowHDR = true;
                cam.clearFlags = CameraClearFlags.Skybox;
                var data = cam.GetUniversalAdditionalCameraData();
                if (data != null) data.renderPostProcessing = true;
            }
        }
    }
}
  • It finds the scene's Directional Light (the sun). If there is none (menu, Game scene), it makes one.
  • It makes a global Volume with Bloom, Vignette, Tonemapping and Color Adjustments. This is the post-processing.
  • The last part calls DayNight.Create(...). night comes from the level (LevelDef.night), so Levels 3 and 4 start at night.
  • GameManager calls EnvironmentSetup.Apply(camera, false, Def.night). The menu calls it with menu = true.

8Script: DayNightButton (and add it to the menu)

Assets/GoldRush/Scripts/UI/DayNightButton.cs – replace the whole file.

using UnityEngine;
using UnityEngine.UI;

namespace GoldRush
{
    /// <summary>Round HUD button that switches day ↔ night in real time (also the N key).
    /// Shows a sun by day and a crescent moon by night.</summary>
    public class DayNightButton : MonoBehaviour
    {
        [SerializeField] RectTransform sun, moon;

        public static DayNightButton Create(Transform parent, Vector2 anchor, Vector2 pos, float size = 100f, bool glass = false)
        {
            float k = size / 100f;
            var root = UIKit.Rect("DayNight", parent).At(anchor, pos, new Vector2(size, size));
            var face = glass ? UIKit.Glass(root, Sprites.Circle) : UIKit.Chip3D(root, Sprites.Circle, UIKit.HudBg);
            var b = root.gameObject.AddComponent<DayNightButton>();

            // sun: disc + 8 rays
            // Material Icons: light_mode (sun) / dark_mode (moon)
            float g = size * 0.56f;
            b.sun = UIKit.IconGraphic(face.transform, UIKit.Icon.Sun, g, UIKit.Gold);
            b.sun.At(new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(g, g), new Vector2(0.5f, 0.5f));
            b.moon = UIKit.IconGraphic(face.transform, UIKit.Icon.Moon, g, UIKit.Cream);
            b.moon.At(new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(g, g), new Vector2(0.5f, 0.5f));

            var btn = root.gameObject.AddComponent<Button>();
            btn.targetGraphic = face;
            b.Refresh(DayNight.IsNight);
            return b;
        }

        void OnChanged(bool night)
        {
            if (!this) return;
            Refresh(night);
            Tween.Punch(transform, 0.2f, 0.3f);
        }

        void Refresh(bool night)
        {
            // the icon shows the current time of day
            sun.gameObject.SetActive(!night);
            moon.gameObject.SetActive(night);
        }

        void Update()
        {
            if (Input.GetKeyDown(KeyCode.N)) { AudioFX.Play("whoosh", 0.7f); DayNight.Toggle(); }
            if (sun.gameObject.activeSelf) sun.localRotation = Quaternion.Euler(0, 0, Time.unscaledTime * 12f);
        }

        void Awake()
        {
            // connected here (not in Create) so it also works when saved in a scene
            var btn = GetComponent<Button>();
            if (btn) btn.onClick.AddListener(() => { AudioFX.Play("whoosh", 0.7f); DayNight.Toggle(); });
        }

        void OnEnable() { DayNight.Changed += OnChanged; }
        void OnDisable() { DayNight.Changed -= OnChanged; }
        void Start() { Refresh(DayNight.IsNight); }
    }
}
  • The game HUD makes this button from code with Create(...) (next to the pause button).
  • In Awake it connects the click. So it also works on a button that you saved in a scene.
  • It listens to DayNight.Changed and shows the sun by day, the moon by night. The sun turns slowly.

Now add one to the menu by hand:

  1. Open MainMenu.unity. Right-click MenuUI/Title → UI → Legacy → Button. Name it DayNight. Delete its Text child.
  2. Anchor top left, pivot 0, 1. Pos 122, -30. Size 76 × 76. Image: circle.png from Resources/UISprites, slate 616B85.
  3. Right-click DayNight → UI → Image. Name it Sun. Size 43 × 43. Source Image ic_light_mode, colour gold FFC233.
  4. Again: UI → Image, name Moon, size 43 × 43, Source Image ic_dark_mode, colour cream F2F7FF.
  5. Select DayNight → Add Component → Day Night Button. Drag Sun into Sun and Moon into Moon.
  6. Add Component → Button FX, tick Flat.
  7. Optional: copy it to the top-right bar (child of MenuUI, anchor top right, pos -362, -34, size 100 × 100, icons 56 × 56). Add the copy to GoldRushMenu → Top Right (size 3). Save the scene.
Do not add a Menu Action to this button. Day Night Button handles the click itself.

9Script: VirtualJoystick (left half)

On a phone there is no keyboard or mouse. The screen is split in two:

  • Left half: touch and drag to move. A stick appears under your thumb. Push it to the edge to sprint (the ring turns orange).
  • Right half: drag to turn the camera.

Both are invisible UI areas that GameUI makes at the bottom of the HUD: TouchLookArea.Create(...), then VirtualJoystick.Create(...). You do not place them in the scene.

Assets/GoldRush/Scripts/UI/VirtualJoystick.cs – replace the whole file.

using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

namespace GoldRush
{
    /// <summary>
    /// Touch movement for the LEFT half of the screen. Invisible until a finger touches down;
    /// then a stick appears under the thumb. Pushing the stick to the outer edge = sprint.
    /// </summary>
    public class VirtualJoystick : MonoBehaviour, IPointerDownHandler, IDragHandler, IPointerUpHandler
    {
        public RectTransform area, stickBase, knob;
        public float radius = 120f;
        public Vector2 Value { get; private set; }
        /// <summary>True while a finger is on the stick.</summary>
        public bool Held => pointer != int.MinValue;

        CanvasGroup group;
        Image edge;
        int pointer = int.MinValue;
        float shown;

        public static VirtualJoystick Create(Transform parent)
        {
            var area = UIKit.Rect("MoveArea (left half)", parent);
            area.anchorMin = Vector2.zero; area.anchorMax = new Vector2(0.5f, 1f);
            area.pivot = Vector2.zero; area.offsetMin = area.offsetMax = Vector2.zero;
            area.gameObject.AddComponent<Image>().color = new Color(0, 0, 0, 0); // invisible touch target

            var b = UIKit.Rect("Stick", area);
            b.anchorMin = b.anchorMax = Vector2.zero;
            b.pivot = new Vector2(0.5f, 0.5f);
            b.sizeDelta = new Vector2(300, 300);
            var bg = UIKit.Img(b, Sprites.Circle, UIKit.WithAlpha(UIKit.HudBg, 0.45f), "Bg"); bg.rectTransform.Fill();
            var ring = UIKit.Img(b, Sprites.Ring, UIKit.WithAlpha(UIKit.Cyan, 0.6f), "Ring"); ring.rectTransform.Fill();
            var k = UIKit.Rect("Knob", b);
            k.At(new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(130, 130));
            var ks = UIKit.Img(k, Sprites.Circle, new Color(0, 0, 0, 0.25f), "Shadow"); ks.rectTransform.Fill(-4, -10, -4, 2);
            var kf = UIKit.Img(k, Sprites.Circle, UIKit.Cream, "Face"); kf.rectTransform.Fill();
            var kg = UIKit.Img(k, Sprites.Circle, UIKit.Cyan, "Dot"); kg.rectTransform.Fill(34, 34, 34, 34);

            var j = area.gameObject.AddComponent<VirtualJoystick>();
            j.area = area; j.stickBase = b; j.knob = k; j.edge = ring;
            j.group = b.gameObject.AddComponent<CanvasGroup>();
            j.group.blocksRaycasts = false;
            j.group.alpha = 0f; // hidden until touched
            return j;
        }

        Vector2 Local(PointerEventData e)
        {
            RectTransformUtility.ScreenPointToLocalPointInRectangle(area, e.position, e.pressEventCamera, out var p);
            return p;
        }

        public void OnPointerDown(PointerEventData e)
        {
            if (Held) return;
            pointer = e.pointerId;
            stickBase.anchoredPosition = Local(e);
            knob.anchoredPosition = Vector2.zero;
            OnDrag(e);
        }

        public void OnDrag(PointerEventData e)
        {
            if (e.pointerId != pointer) return;
            Vector2 d = Local(e) - stickBase.anchoredPosition;
            // the stick follows the thumb if it drags far away
            if (d.magnitude > radius * 1.6f)
            {
                stickBase.anchoredPosition += d.normalized * (d.magnitude - radius * 1.6f);
                d = Local(e) - stickBase.anchoredPosition;
            }
            Vector2 c = Vector2.ClampMagnitude(d, radius);
            knob.anchoredPosition = c;
            Vector2 v = c / radius;
            Value = v.magnitude < 0.12f ? Vector2.zero : v;
        }

        public void OnPointerUp(PointerEventData e)
        {
            if (e.pointerId != pointer) return;
            Release();
        }

        public void Release()
        {
            pointer = int.MinValue;
            Value = Vector2.zero;
            if (knob) knob.anchoredPosition = Vector2.zero;
        }

        void Update()
        {
            shown = Mathf.MoveTowards(shown, Held ? 1f : 0f, Time.unscaledDeltaTime * 8f);
            group.alpha = shown;
            // the outer ring lights up orange while sprinting (stick pushed to the edge)
            if (edge) edge.color = Held && Value.magnitude > 0.9f ? UIKit.WithAlpha(UIKit.Orange, 0.9f) : UIKit.WithAlpha(UIKit.Cyan, 0.6f);
        }

        void OnDisable() { Release(); }
    }
}
  • IPointerDownHandler, IDragHandler, IPointerUpHandler: Unity UI calls these when a finger (or mouse) goes down, moves and goes up on the area.
  • pointer remembers which finger owns the stick, so a second finger on the right half does not move it.
  • Value is the stick direction, from -1 to 1. Very small values (under 0.12) count as zero.
  • PlayerMotor reads joystick.Value to move, and sprints when Value.magnitude > 0.9.

10Script: TouchLookArea (right half)

Assets/GoldRush/Scripts/UI/TouchLookArea.cs – replace the whole file.

using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

namespace GoldRush
{
    /// <summary>Touch camera control for the RIGHT half of the screen: drag to look around.
    /// A small ring appears under the finger while dragging.</summary>
    public class TouchLookArea : MonoBehaviour, IDragHandler, IPointerDownHandler, IPointerUpHandler
    {
        static Vector2 accumulated;
        int pointer = int.MinValue;
        RectTransform area, marker;
        CanvasGroup markerGroup;

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
        static void ResetStatics() { accumulated = Vector2.zero; }

        /// <summary>Returns and clears the drag delta (screen pixels) since the last call.</summary>
        public static Vector2 Consume()
        {
            var a = accumulated;
            accumulated = Vector2.zero;
            return a;
        }

        public static TouchLookArea Create(Transform parent)
        {
            var rt = UIKit.Rect("LookArea (right half)", parent);
            rt.anchorMin = new Vector2(0.5f, 0f);
            rt.anchorMax = new Vector2(1f, 1f);
            rt.pivot = Vector2.zero;
            rt.offsetMin = rt.offsetMax = Vector2.zero;
            rt.gameObject.AddComponent<Image>().color = new Color(0, 0, 0, 0);
            var t = rt.gameObject.AddComponent<TouchLookArea>();
            t.area = rt;
            t.marker = UIKit.Rect("Finger", rt);
            t.marker.anchorMin = t.marker.anchorMax = Vector2.zero;
            t.marker.sizeDelta = new Vector2(150, 150);
            UIKit.Img(t.marker, Sprites.Ring, UIKit.WithAlpha(UIKit.Cyan, 0.5f), "Ring").rectTransform.Fill();
            UIKit.Img(t.marker, Sprites.Soft, UIKit.WithAlpha(UIKit.Cyan, 0.25f), "Glow").rectTransform.Fill(-30, -30, -30, -30);
            t.markerGroup = t.marker.gameObject.AddComponent<CanvasGroup>();
            t.markerGroup.alpha = 0f;
            t.markerGroup.blocksRaycasts = false;
            return t;
        }

        void Place(PointerEventData e)
        {
            RectTransformUtility.ScreenPointToLocalPointInRectangle(area, e.position, e.pressEventCamera, out var p);
            marker.anchoredPosition = p;
        }

        public void OnPointerDown(PointerEventData e)
        {
            if (pointer != int.MinValue) return;
            pointer = e.pointerId;
            Place(e);
        }

        public void OnPointerUp(PointerEventData e) { if (e.pointerId == pointer) pointer = int.MinValue; }

        public void OnDrag(PointerEventData e)
        {
            if (e.pointerId != pointer) return;
            accumulated += e.delta;
            Place(e);
        }

        void Update()
        {
            if (markerGroup) markerGroup.alpha = Mathf.MoveTowards(markerGroup.alpha, pointer != int.MinValue ? 1f : 0f, Time.unscaledDeltaTime * 8f);
        }

        void OnDisable() { pointer = int.MinValue; }
    }
}
  • OnDrag adds the finger movement (e.delta, in pixels) to accumulated.
  • Consume() gives back the movement since last time and resets it to zero.
  • ThirdPersonCamera calls TouchLookArea.Consume() in LateUpdate and turns the camera with it (touchSensitivity = 0.16).

12Script: AudioFX (music)

The final AudioFX can make its sounds in code when the .wav files are missing. That code is in a helper. Drag Synth.cs from GoldRush_Export/Resources/Scripts/Helpers/Audio/ into Assets/GoldRush/Scripts/Audio/ first.

Assets/GoldRush/Scripts/Audio/AudioFX.cs – replace the whole file with this final version.

using System.Collections.Generic;
using UnityEngine;

namespace GoldRush
{
    /// <summary>Global audio player for synthesized sound effects and background music.</summary>
    public class AudioFX : MonoBehaviour
    {
        const int PoolSize = 8;
        const float MusicVolume = 0.3f;
        const float TensionPitch = 1.07f;
        const float TensionVolume = 1.15f;

        static AudioFX inst;
        static bool quitting;

        Dictionary<string, AudioClip> clips;
        AudioSource[] pool;
        int next;
        AudioSource music;

        bool tension;
        float tensionVolCur = 1f;
        float duckAmount;
        float duckTimer;
        float duckCur;

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
        static void ResetStatics()
        {
            inst = null;
            quitting = false;
        }

        static AudioFX I
        {
            get
            {
                if (inst != null) return inst;
                if (quitting || !Application.isPlaying) return null;
                var go = new GameObject("[AudioFX]");
                DontDestroyOnLoad(go);
                go.AddComponent<AudioFX>(); // Awake assigns inst
                return inst;
            }
        }

        void Awake()
        {
            if (inst != null && inst != this)
            {
                Destroy(gameObject);
                return;
            }
            inst = this;
            Init();
        }

        void Init()
        {
            if (clips != null) return;
            // the sound files (Art/Audio/*.wav) are listed in the GoldRushAssets asset; "coin.wav" -> id "coin"
            clips = new Dictionary<string, AudioClip>();
            var assets = GameAssets.I;
            if (assets && assets.sounds != null)
                foreach (var clip in assets.sounds)
                    if (clip) clips[clip.name] = clip;
            if (clips.Count == 0) clips = Synth.BuildAll(); // no sound files yet: make them from code

            pool = new AudioSource[PoolSize];
            for (int i = 0; i < PoolSize; i++)
            {
                var s = gameObject.AddComponent<AudioSource>();
                s.playOnAwake = false;
                s.loop = false;
                s.spatialBlend = 0f;
                s.priority = 64;
                pool[i] = s;
            }

            music = gameObject.AddComponent<AudioSource>();
            music.playOnAwake = false;
            music.loop = true;
            music.spatialBlend = 0f;
            music.priority = 0;
            music.volume = MusicVolume;
            music.mute = !Save.Data.music;
        }

        void OnApplicationQuit() { quitting = true; }

        void OnDestroy()
        {
            if (inst == this) inst = null;
        }

        // ------------------------------------------------------------------ public API

        public static void Play(string id, float volume = 1f, float pitch = 1f)
        {
            if (!Save.Data.sfx || string.IsNullOrEmpty(id)) return;
            var i = I;
            if (i == null) return;
            i.PlayInternal(id, volume, pitch);
        }

        public static void StartMusic()
        {
            var i = I;
            if (i == null) return;
            i.music.mute = !Save.Data.music;
            if (i.music.isPlaying) return;
            AudioClip clip;
            if (!i.clips.TryGetValue("music", out clip) || clip == null) return;
            i.music.clip = clip;
            i.music.loop = true;
            i.music.pitch = i.tension ? TensionPitch : 1f;
            i.music.volume = i.ComputeMusicVolume();
            i.music.Play();
        }

        public static void RefreshMute()
        {
            var i = I;
            if (i == null) return;
            i.music.mute = !Save.Data.music;
            if (!Save.Data.sfx)
            {
                for (int k = 0; k < i.pool.Length; k++)
                    if (i.pool[k] != null && i.pool[k].isPlaying) i.pool[k].Stop();
            }
        }

        public static void SetTension(bool chased)
        {
            var i = I;
            if (i == null) return;
            i.tension = chased;
        }

        public static void Duck(float amount, float seconds)
        {
            var i = I;
            if (i == null) return;
            amount = Mathf.Clamp01(amount);
            if (i.duckTimer > 0f) i.duckAmount = Mathf.Max(i.duckAmount, amount);
            else i.duckAmount = amount;
            i.duckTimer = Mathf.Max(i.duckTimer, seconds);
        }

        // ------------------------------------------------------------------ internals

        void PlayInternal(string id, float volume, float pitch)
        {
            AudioClip clip;
            if (clips == null || !clips.TryGetValue(id, out clip) || clip == null)
            {
                Debug.LogWarning("[AudioFX] Unknown sound id: " + id);
                return;
            }

            AudioSource src = null;
            for (int k = 0; k < PoolSize; k++)
            {
                var s = pool[(next + k) % PoolSize];
                if (!s.isPlaying) { src = s; next = (next + k + 1) % PoolSize; break; }
            }
            if (src == null)
            {
                src = pool[next];
                next = (next + 1) % PoolSize;
            }

            src.Stop();
            src.clip = clip;
            src.volume = Mathf.Clamp01(volume);
            src.pitch = Mathf.Clamp(pitch, 0.1f, 3f);
            src.Play();
        }

        float ComputeMusicVolume()
        {
            return MusicVolume * tensionVolCur * (1f - duckCur);
        }

        void Update()
        {
            if (music == null) return;
            float dt = Time.unscaledDeltaTime;

            float targetPitch = tension ? TensionPitch : 1f;
            float targetVol = tension ? TensionVolume : 1f;
            float k = 1f - Mathf.Exp(-dt * 1.5f);
            music.pitch = Mathf.Lerp(music.pitch, targetPitch, k);
            if (Mathf.Abs(music.pitch - targetPitch) < 0.0005f) music.pitch = targetPitch;
            tensionVolCur = Mathf.Lerp(tensionVolCur, targetVol, k);

            if (duckTimer > 0f) duckTimer -= dt;
            float targetDuck = duckTimer > 0f ? duckAmount : 0f;
            float rate = targetDuck > duckCur ? 10f : 2f;
            duckCur = Mathf.Lerp(duckCur, targetDuck, 1f - Mathf.Exp(-dt * rate));
            if (targetDuck == 0f && duckCur < 0.001f) { duckCur = 0f; duckAmount = 0f; }

            music.volume = ComputeMusicVolume();
        }
    }
}
  • StartMusic() plays the music clip on a loop. The menu and every level call it. If it already plays, nothing happens, so the music does not restart when the scene changes.
  • SetTension(true): a monster is chasing you. The music gets a bit faster (pitch 1.07) and louder.
  • Duck(amount, seconds) makes the music quieter for a moment, so you hear the win or lose sound.
  • RefreshMute() reads Save.Data.music and Save.Data.sfx. The menu Settings and the pause menu call it.
  • Music volume is 0.3. Eight AudioSources play sound effects, so several sounds can play at once.
  • The sounds still come from the Sounds list of GoldRushAssets. Only if that list is empty, Synth.BuildAll() makes every sound in code (a fallback).
